name: Push to ACR

on:
  push:
    branches: [ "main" ]
  workflow_dispatch: # Allows you to run this button manually from the Actions tab

jobs:
  build-and-push:
    runs-on: ubuntu-latest # This runner has internet access!

    steps:
    - name: Checkout code
      uses: actions/checkout@v3

    - name: Log in to Azure Container Registry
      uses: docker/login-action@v2
      with:
        registry: ${{ secrets.ACR_LOGIN_SERVER }}
        username: ${{ secrets.ACR_USERNAME }}
        password: ${{ secrets.ACR_PASSWORD }}

    - name: Build and Push Docker Image
      uses: docker/build-push-action@v4
      with:
        context: .
        push: true
        # Tags the image with 'latest' and the git commit SHA for versioning
        tags: |
          ${{ secrets.ACR_LOGIN_SERVER }}/forgerock-am:latest
          ${{ secrets.ACR_LOGIN_SERVER }}/forgerock-am:${{ github.sha }}
