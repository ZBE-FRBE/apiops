name: Build and push image to ACR

on:
  push:
    branches: [ "main" ]
  workflow_dispatch:

env:
  IMAGE_NAME: forgerock-ig
  IG_DIR: identity-gateway-2025.11.0

jobs:
  build-and-push:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Set up Docker Buildx
        uses: docker/setup-buildx-action@v3

      - name: Login to ACR
        uses: docker/login-action@v3
        with:
          registry: ${{ secrets.ACR_LOGIN_SERVER }}
          username: ${{ secrets.ACR_USERNAME }}
          password: ${{ secrets.ACR_PASSWORD }}

      - name: Build and push
        uses: docker/build-push-action@v6
        with:
          context: ./${{ env.IG_DIR }}
          file: ./${{ env.IG_DIR }}/docker/Dockerfile
          push: true
          tags: |
            ${{ secrets.ACR_LOGIN_SERVER }}/${{ env.IMAGE_NAME }}:${{ github.sha }}
            ${{ secrets.ACR_LOGIN_SERVER }}/${{ env.IMAGE_NAME }}:latest

FROM gcr.io/forgerock-io/java-21:latest

ENV INSTALL_DIR=/opt/ig
ENV IG_INSTANCE_DIR=/var/ig

COPY --chown=forgerock:root . "${INSTALL_DIR}"

RUN chmod 755 "${INSTALL_DIR}/bin/start.sh" "${INSTALL_DIR}/bin/stop.sh" \
 && mkdir -p "${IG_INSTANCE_DIR}" \
 && chown -R forgerock:root "${IG_INSTANCE_DIR}" "${INSTALL_DIR}" \
 && chmod -R u+rwX,g+rwX "${IG_INSTANCE_DIR}" "${INSTALL_DIR}"

USER 11111

# exec-form = cleaner in containers (signals, no extra shell)
ENTRYPOINT ["/opt/ig/bin/start.sh", "/var/ig"]
